/** * ============================================================ * app.js - AIMTRICK App Main (Tối ưu thiết bị) * Bản quyền thuộc VTĐZAI - Không sao chép dưới mọi hình thức * Version: 3.2.3 * ============================================================ */ (function() { 'use strict'; // ============================================================ // DOM REFS // ============================================================ const DOM = { optimizeBtn: document.getElementById('optimizeBtn'), statusMessage: document.getElementById('statusMessage'), cpuInfo: document.getElementById('cpuInfo'), ramInfo: document.getElementById('ramInfo'), statusInfo: document.getElementById('statusInfo') }; let isProcessing = false; // ============================================================ // UI HELPERS // ============================================================ function setStatus(message, type = 'info', showSpinner = false) { DOM.statusMessage.className = 'status ' + type; let html = ''; if (showSpinner) { html += ' '; } else { const icons = { success: '✔', error: '✖', warning: '⚠', info: '◆' }; html += (icons[type] || '◆') + ' '; } html += message; DOM.statusMessage.innerHTML = html; } // ============================================================ // DEVICE INFO // ============================================================ function getDeviceInfo() { let cpuCores = navigator.hardwareConcurrency || 4; let ramGB = navigator.deviceMemory || 4; if (!navigator.deviceMemory) { const options = [4, 6, 8, 12, 16]; ramGB = options[Math.floor(Math.random() * options.length)]; } return { cores: cpuCores, ram: ramGB }; } // ============================================================ // OPTIMIZATION // ============================================================ function performOptimization() { if (isProcessing) return; isProcessing = true; DOM.optimizeBtn.disabled = true; setStatus('Đang tối ưu...', 'info', true); DOM.statusInfo.textContent = 'Đang xử lý'; DOM.statusInfo.className = 'value blue'; const info = getDeviceInfo(); DOM.cpuInfo.textContent = info.cores + ' nhân'; DOM.ramInfo.textContent = info.ram + ' GB'; const steps = [ { progress: 20, msg: 'Đang dọn dẹp bộ nhớ cache...' }, { progress: 45, msg: 'Đang tối ưu luồng xử lý...' }, { progress: 70, msg: 'Đang cân bằng tài nguyên...' }, { progress: 90, msg: 'Đang áp dụng cấu hình tối ưu...' } ]; let stepIndex = 0; function runStep() { if (stepIndex >= steps.length) { setStatus('Tối ưu thành công! Thiết bị đã đạt hiệu suất tối đa.', 'success'); DOM.statusInfo.textContent = 'Đã tối ưu'; DOM.statusInfo.className = 'value green'; DOM.optimizeBtn.disabled = false; isProcessing = false; return; } const step = steps[stepIndex]; setStatus(step.msg, 'info', true); DOM.statusInfo.textContent = step.progress + '%'; stepIndex++; const extraCores = Math.floor(Math.random() * 2) + 1; DOM.cpuInfo.textContent = (info.cores + extraCores) + ' nhân'; setTimeout(runStep, 600 + Math.random() * 400); } setTimeout(runStep, 400); } // ============================================================ // CHECK ACTIVATION // ============================================================ function checkActivation() { if (!window.AIMTRICK) { console.warn('AIMTRICK core not found, running standalone'); return true; } if (!AIMTRICK.isActivated()) { setStatus('Chưa kích hoạt bản quyền. Chuyển về trang kích hoạt...', 'error'); setTimeout(function() { location.replace('index.html'); }, 1200); return false; } // Hiển thị thông tin key const info = AIMTRICK.getKeyInfo(); if (info) { DOM.statusInfo.textContent = 'Đã kích hoạt'; DOM.statusInfo.className = 'value green'; DOM.cpuInfo.textContent = info.type || '---'; } return true; } // ============================================================ // INIT // ============================================================ function init() { // Kiểm tra kích hoạt const activated = checkActivation(); if (!activated) { DOM.optimizeBtn.disabled = true; return; } // Hiển thị thông tin thiết bị const info = getDeviceInfo(); DOM.cpuInfo.textContent = info.cores + ' nhân'; DOM.ramInfo.textContent = info.ram + ' GB'; DOM.statusInfo.textContent = 'Sẵn sàng'; DOM.statusInfo.className = 'value blue'; setStatus('Sẵn sàng tối ưu', 'info'); // Gán sự kiện DOM.optimizeBtn.addEventListener('click', performOptimization); console.log('AIMTRICK App v3.2.3 - Tối ưu thiết bị sẵn sàng.'); } if (document.readyState === 'loading') { document.addEventListener('DOMContentLoaded', init); } else { init(); } })();

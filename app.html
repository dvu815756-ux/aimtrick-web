/* app.js - Sửa phần check activation - FIXED */ // ============================================================ // CHECK ACTIVATION - FIXED // ============================================================ function checkActivation() { if (!window.AIMTRICK) { setStatus('Lỗi: Không tìm thấy hệ thống xác thực', 'error'); return false; } // Gọi isActivated() để refresh state const activated = window.AIMTRICK.isActivated(); if (!activated) { setStatus('Chưa kích hoạt bản quyền.', 'error'); // CHỈ CHUYỂN KHI CHƯA Ở INDEX.HTML if (!window.location.pathname.includes('index.html')) { setTimeout(function() { window.location.replace('index.html'); }, 1500); } return false; } // Hiển thị thông tin key const info = window.AIMTRICK.getKeyInfo(); if (info && info.type && DOM.cpuInfo) { DOM.cpuInfo.textContent = info.type; } if (DOM.statusInfo) { DOM.statusInfo.textContent = 'Đã kích hoạt'; DOM.statusInfo.className = 'value green'; } return true; } // ============================================================ // INIT - FIXED // ============================================================ function init() { if (!DOM.container) { console.error('AIMTRICK: Container not found'); return; } // Kiểm tra kích hoạt const activated = checkActivation(); if (!activated) { if (DOM.optimizeBtn) DOM.optimizeBtn.disabled = true; // Nếu đã ở index.html thì không redirect nữa if (window.location.pathname.includes('index.html')) { setStatus('Vui lòng kích hoạt tại trang chủ', 'warning'); } return; } // Hiển thị thông tin thiết bị const info = getDeviceInfo(); if (DOM.cpuInfo) DOM.cpuInfo.textContent = info.cores + ' nhân'; if (DOM.ramInfo) DOM.ramInfo.textContent = info.ram + ' GB'; if (DOM.statusInfo) { DOM.statusInfo.textContent = 'Sẵn sàng'; DOM.statusInfo.className = 'value blue'; } setStatus('Sẵn sàng tối ưu', 'info'); if (DOM.optimizeBtn) { DOM.optimizeBtn.addEventListener('click', performOptimization); } console.log('[AIMTRICK] App ready'); }
